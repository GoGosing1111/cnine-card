import fs from 'node:fs/promises';
import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
import {connect,safe} from '../rap-contest-rewards-20261027/preflight.mjs';

const db=connect();try{

  targets.push({code,currentLimit:policy.stockLimit,crafted,addition:10,scheduledLimit:policy.stockLimit+10});
 }
 for(const old of receipt.before.packMeta)assert.equal(await value(old.key),old.value);
 const audit=(await db.query("SELECT id::text,created_at,after_data FROM admin_logs WHERE action_type='MERCENARY_CRAFT_STOCK_SCHEDULE' AND after_data::jsonb->'schedule'->>'id'=$1",[ID])).rows;assert.equal(audit.length,1);assert.equal(audit[0].id,receipt.auditId);
 assert.equal(Number((await db.query("SELECT COUNT(*) n FROM admin_logs WHERE action_type='MERCENARY_CRAFT_STOCK_SCHEDULE_APPLY' AND target_id=$1",[ID])).rows[0].n),0);
 const alarm=JSON.parse(await value('mercenary_stock_alarm_v1'));assert.equal(alarm.nextBySource.limitedCraft,AT);assert.equal(alarm.nextAt,'2026-11-05T11:00:00.000Z');assert.equal(alarm.nextAlarmAt,Date.parse(alarm.nextAt));assert(Date.parse(alarm.checkedAt)>=Date.parse(receipt.registeredAt));
 const writes=Number((await db.query('SELECT SUM(n_tup_ins+n_tup_upd+n_tup_del) n FROM pg_stat_xact_user_tables')).rows[0].n);assert.equal(writes,0);
 await db.query('ROLLBACK');const result={status:'INDEPENDENTLY_VERIFIED_SCHEDULED',operationKey:OP,readOnly,productionWrites:0,registeredAt:receipt.registeredAt,auditId:receipt.auditId,schedule,targets,hiddenRecipesPreserved:8,packSchedulesPreserved:4,policiesSha256:createHash('sha256').update(JSON.stringify(policies)).digest('hex'),alarm,appliedCount:0,checkedAt:new Date().toISOString()};
 await fs.writeFile(new URL('evidence/independent.json',import.meta.url),JSON.stringify(result,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify(result));
